Spark Structured Streaming
# ===== 位置：/home/hadoop/project/spark_write_mysql.py (PySpark) =====
from pyspark.sql import SparkSession
from pyspark.sql.functions import (from_json, col, window,
                                   avg, max, current_timestamp)

spark = SparkSession.builder.appName("GreenhouseStreamAgg").getOrCreate()

# 1) Kafka Source：订阅 sensor_data
df = spark.readStream \
    .format("kafka") \
    .option("kafka.bootstrap.servers", "localhost:9092") \
    .option("subscribe", "sensor_data") \
    .load()

# 2) JSON 解析 -> 提取大棚编号、温湿度、光照、时间戳
json_df = df.select(
    from_json(col("value").cast("string"), schema).alias("data")
).select("data.*")

# 3) 处理时间：用 Spark 读到消息的时间作为事件时间
#    （与 Flink 用数据自带 timestamp 的事件时间形成对比）
df_with_ts = json_df.withColumn("event_time", current_timestamp())

# 4) 5 秒滚动窗口 + 10 秒 Watermark（允许晚到数据）
agg_df = df_with_ts \
    .withWatermark("event_time", "10 seconds") \
    .groupBy(col("greenhouse_id"),
             window(col("event_time"), "5 seconds")) \
    .agg(avg("air_temp").alias("avg_temp"),
         avg("air_humidity").alias("avg_humidity"),
         max("light").alias("max_light"))

# 5) Sink：foreachBatch 写入 MySQL window_stat
def write_to_mysql(batch_df, batch_id):
    batch_df.write.mode("append").jdbc(
        url=jdbc_url, table="window_stat", properties=props)

agg_df.writeStream \
    .outputMode("append") \
    .foreachBatch(write_to_mysql) \
    .option("checkpointLocation", "/tmp/greenhouse_chk_v3") \
    .start()
Flink DataStream API
# ===== 位置：/home/hadoop/flink_project/flink_window_stat.py (PyFlink) =====
from pyflink.datastream import StreamExecutionEnvironment
from pyflink.datastream.connectors import FlinkKafkaConsumer, JdbcSink
from pyflink.common import WatermarkStrategy, Duration, Time
from pyflink.datastream.checkpointing_mode import CheckpointingMode

env = StreamExecutionEnvironment.get_execution_environment()
# 10 秒一次 Checkpoint，EXACTLY_ONCE（位点 + 窗口状态一起持久化）
env.enable_checkpointing(10000, CheckpointingMode.EXACTLY_ONCE)

# 1) Kafka Source：订阅 sensor_data
kafka_consumer = FlinkKafkaConsumer(
    "sensor_data", SimpleStringSchema(),
    {"bootstrap.servers": "127.0.0.1:9092"})

# 2) 事件时间 + Watermark：从数据自带 timestamp 提取，允许 10 秒乱序
watermark_strategy = (
    WatermarkStrategy
    .for_bounded_out_of_orderness(Duration.of_seconds(10))
    .with_timestamp_assigner(SensorTimestampAssigner()))

# 3) 解析 -> 事件时间水印 -> 5 秒事件时间滚动窗口 -> 截断均值聚合
result_stream = (
    env.add_source(kafka_consumer)
       .map(ParseSensorMapFunction())
       .assign_timestamps_and_watermarks(watermark_strategy)
       .key_by(lambda e: e.greenhouse_id)
       .window(TumblingEventTimeWindows.of(Time.seconds(5)))
       .process(GreenhouseWindowFunction())  # 去首尾各1条后取均值，抗噪声
)

# 4) Sink：JDBC 写入 MySQL window_stat_flink（每窗口 1 行）
result_stream.add_sink(JdbcSink.sink(
    "INSERT INTO window_stat_flink "
    "(greenhouse_id, win_start, win_end, avg_temp, "
    " avg_humidity, avg_soil_humidity, max_light) "
    "VALUES (?, ?, ?, ?, ?, ?, ?)",
    OUTPUT_TYPE, jdbc_conn_opts, jdbc_exec_opts))

env.execute("greenhouse_flink_window_stat")

📌 双引擎核心差异（本项目真实实现）： Spark 用 处理时间（current_timestamp()）做窗口，内置 avg/max 算术聚合，声明式 SQL DSL； Flink 用 数据自带事件时间 + Watermark，自定义 ProcessWindowFunction 做截断均值（去首尾各 1 条抗噪声），编程式更灵活。
